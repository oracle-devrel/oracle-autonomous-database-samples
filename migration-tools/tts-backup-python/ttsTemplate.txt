get_tablespaces:
  select tablespace_name from dba_tablespaces where contents = 'PERMANENT' and
  tablespace_name not in ('SYSTEM', 'SYSAUX');

validate_common_user_remap_source:
  declare
    l_msg varchar2(30000);
  begin
    for rec in (
      select t.owner, t.table_name
      from dba_tables t
      join dba_objects o
        on o.owner = t.owner
       and o.object_name = t.table_name
       and o.object_type = 'TABLE'
      where t.owner = '${schema}'
        and t.tablespace_name in (${ts_list})
        and o.oracle_maintained = 'Y'
        ${exc_tbl_filter_t}
    ) loop
      l_msg := l_msg || chr(10) || '  ' || rec.owner || '.' || rec.table_name;
    end loop;

    if l_msg is not null then
      l_msg := 'ERROR : Common-user remap source ${schema} owns Oracle-maintained tables '
            || 'in the selected transport tablespaces. Remapping Oracle-maintained table data '
            || 'is not supported. Move these tables out of the selected tablespaces or add '
            || 'them to EXCLUDE_TABLES before retrying. Objects found:' || l_msg;

      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;
  end;
  /

validate_schemas:
  declare
    l_cnt number := 0;
    l_msg varchar2(30000);
  begin
    select count(*) into l_cnt from dba_users where username = '${schema}';
    if l_cnt = 0 then
      l_msg := 'ERROR : Schema ${schema} not found in the database.';
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else 
        raise_application_error(-20001, l_msg);
      end if;
    end if;

    -- check if there are any virtual index
    l_msg := '';
    for rec in (
      select owner, object_name from dba_objects o
      where o.owner = '${schema}' and o.object_type = 'INDEX' and o.status = 'VALID' and
      not exists (
        select 1 from dba_indexes i
        where i.owner = o.owner and i.index_name = o.object_name
      )
    ) loop
      l_msg := l_msg || chr(10) || rec.owner || '.' || rec.object_name;
    end loop;

    if length(l_msg) > 0 then
      l_msg := 'ERROR : Validation failed for schema ${schema}: The schema contains virtual indexes. ' ||
               'Drop or recreate these indexes as regular indexes before retrying. Virtual indexes:' ||
               l_msg;
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;
  end;
  /

  end;

get_xml_export_tables:
  WITH xml_objs AS (
    SELECT c.owner, t.table_name
    FROM dba_tables t
    JOIN dba_tab_columns c
      ON c.owner = t.owner
    AND c.table_name = t.table_name
    WHERE t.tablespace_name IN (${ts_list}) ${exc_tbl_filter}
      AND c.data_type = 'XMLTYPE'
    UNION
    SELECT c.owner, c.table_name
    FROM dba_segments s
    JOIN dba_tab_columns c
      ON c.owner = s.owner
    AND c.table_name = s.segment_name
    WHERE s.tablespace_name IN (${ts_list}) ${exc_tbl_filter_seg}
      AND c.data_type = 'XMLTYPE'
  ),
  xml_names AS (
    SELECT DISTINCT table_name
    FROM xml_objs
  ),
  same_name_tables AS (
    SELECT DISTINCT t.owner, t.table_name
    FROM dba_tables t
    JOIN xml_names x
      ON x.table_name = t.table_name
    WHERE t.tablespace_name IN (${ts_list}) ${exc_tbl_filter}
    UNION
    SELECT DISTINCT s.owner, s.segment_name AS table_name
    FROM dba_segments s
    JOIN xml_names x
      ON x.table_name = s.segment_name
    WHERE s.tablespace_name IN (${ts_list})  ${exc_tbl_filter_t}
  )
  SELECT owner || '."' || replace(table_name, '"', '""') || '"'
  FROM (
    SELECT DISTINCT owner, table_name
    FROM xml_objs
    UNION
    SELECT DISTINCT s.owner, s.table_name
    FROM same_name_tables s
    JOIN xml_objs x
      ON x.table_name = s.table_name
    AND x.owner <> s.owner
  )
  ORDER BY 1;

get_xml_index_path_tables:
  WITH xml_index_tables AS (
    SELECT x.table_owner AS owner,
          x.path_table_name AS table_name
    FROM   dba_xml_indexes x
    WHERE  x.table_owner IN (${sc_list})
      AND  upper(x.index_type) IN ('UNSTRUCTURED', 'STRUCTURED AND UNSTRUCTURED')
      AND  x.path_table_name IS NOT NULL
    UNION ALL
    -- structured / hybrid XMLTable names
    SELECT x.table_owner AS owner,
          xt.xmltab_name AS table_name
    FROM   dba_xml_indexes x,
          XMLTABLE(
            '/parameters/structured/group/xmltab'
            PASSING x.parameters
            COLUMNS xmltab_name VARCHAR2(128) PATH '@name'
          ) xt
    WHERE  x.table_owner IN (${sc_list})
      AND  upper(x.index_type) IN ('STRUCTURED', 'STRUCTURED AND UNSTRUCTURED')
      AND  x.parameters IS NOT NULL
      AND  xt.xmltab_name IS NOT NULL
  )
  SELECT DISTINCT
         x.table_name AS path_table
  FROM xml_index_tables x
  JOIN dba_tables t
    ON t.owner = x.owner
   AND t.table_name = x.table_name
  WHERE t.tablespace_name IN (${ts_list})
  ORDER BY 1;

validate_tablespaces:
  declare
    l_platform_name v$database.platform_name%TYPE;
    l_ts_block_size dba_tablespaces.block_size%TYPE;
    l_ts_status dba_tablespaces.status%TYPE;
    l_ts_bigfile dba_tablespaces.bigfile%TYPE;
    l_ts_encrypted dba_tablespaces.encrypted%TYPE;
    l_default_ts_name user_users.default_tablespace%TYPE;
    l_cnt number;
    l_nls_charset database_properties.property_value%TYPE;
    l_nls_ncharset database_properties.property_value%TYPE;
    l_timezone varchar2(32);
    l_tts_charset_status boolean;
    l_error_msg varchar2(1024);
    l_msg varchar2(30000);
  begin
    if ('${tablespace}' = 'SYSTEM' or '${tablespace}' = 'SYSAUX') then
      l_msg := 'ERROR : Administrative tablespaces SYSTEM, SYSAUX cannot be transported';
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;

    -- Check if platform is supported for transport
    begin
      select d.platform_name into l_platform_name
      from v$database d, v$transportable_platform t
      where d.platform_id = t.platform_id;
    exception
      when no_data_found then
        l_msg := 'ERROR : Source database platform ' || l_platform_name || ' is not supported for transportable tablespace.';
        if ('${dry_run}' = 'TRUE') then
          dbms_output.put_line(l_msg);
        else
          raise_application_error(-20001, l_msg);
        end if;
    end;

    -- Check tablespace properties
    select block_size, bigfile, encrypted, status 
    into l_ts_block_size, l_ts_bigfile, l_ts_encrypted, l_ts_status 
    from dba_tablespaces where tablespace_name = '${tablespace}';

    if (l_ts_block_size != 8192) then
      l_msg := 'ERROR : Tablespace ${tablespace} has block size ' || l_ts_block_size || '. Only 8K block size is supported.';
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;

    if ('${final_backup}' = 'TRUE' and l_ts_status != 'READ ONLY') then
      l_msg := 'ERROR : Tablespace ${tablespace} is not in read only mode in final backup';
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;

    -- check if the tablespace is default tablespace of current user
    -- check if current user's default tablespace is in read write mode
    select u.default_tablespace, t.status into l_default_ts_name, l_ts_status
    from user_users u, dba_tablespaces t where u.default_tablespace = t.tablespace_name;
    if (l_default_ts_name = '${tablespace}') then
      l_msg := 'ERROR : Tablespace ${tablespace} can not be default tablespace of user running transport.';
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;
    if (l_ts_status != 'ONLINE') then
      l_msg := 'ERROR : Default tablespace ' || l_default_ts_name || ' of current user is not in READ WRITE mode';
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;

    -- check if any of the table column data type owners are not in schema list
    l_msg := '';
    for rec in (
      select atc.owner,atc.table_name,atc.column_name,atc.data_type_owner,atc.data_type from all_tab_columns atc 
      JOIN all_tables at ON atc.table_name = at.table_name and atc.owner = at.owner 
      and at.tablespace_name = '${tablespace}' and atc.data_type_owner NOT IN (${sc_list})
      ${exc_tbl_filter_dt} and atc.data_type <> 'XMLTYPE' and atc.owner IN (${sc_list})
    order by atc.owner, atc.table_name, atc.column_name
    ) loop
      l_msg := l_msg || chr(10) ||
               rec.owner || '.' || rec.table_name || '.' || rec.column_name ||
               ' references type owner ' || rec.data_type_owner ||
               ' (' || rec.data_type || ')';
    end loop;  
    
    if length(l_msg) > 0 then
      l_msg := 
        'Validation failed for tablespace "${tablespace}": Some columns reference data types owned by schemas that are not in the allowed transport list. ' || 
        'Ensure all referenced data types belong to the transported schemas. ' ||
        'Failing columns:' || l_msg;
      if ('${ignore_non_fatal_errors}' = 'TRUE') then
        dbms_output.put_line('WARNING : ' || l_msg);
        dbms_output.put_line(
          'WARNING : IGNORE_NON_FATAL_ERRORS is set to TRUE. ' ||
          'This validation error is being ignored. The referenced data type owner/schema ' ||
          'is outside the transported schema list. Restore/import at ADBS may report ' ||
          'errors if the referenced type or owner is unavailable at the target; existing ' ||
          'compatible objects at ADBS may allow the import to succeed.');
      elsif ('${dry_run}' = 'TRUE') then
        dbms_output.put_line('ERROR : ' || l_msg);
      else
        raise_application_error(-20001, 'ERROR : ' || l_msg);
      end if;
    end if;

    -- check if xml type tables of storage OBJECT-RELATIONAL or CLOB is used by the tablespace
    -- These storage types are not supported by target
    l_msg := '';
    for rec in (
      SELECT DISTINCT s.owner, x.table_name, x.storage_type, nvl(x.nonschema, 'NO') as nonschema
      FROM dba_segments s
      JOIN dba_xml_tab_cols x
      ON s.owner = x.owner
      AND s.segment_name = x.table_name
      JOIN dba_tab_cols c
      ON c.owner = x.owner
      AND c.table_name = x.table_name
      AND c.column_name = x.column_name
      WHERE s.tablespace_name = '${tablespace}' ${exc_tbl_filter_xml}
      and c.hidden_column = 'NO'
      and ( x.storage_type <> 'BINARY' or nvl(x.nonschema, 'NO') <> 'YES' )
      UNION
      SELECT DISTINCT s.owner, xm.table_name, xm.storage_type, nvl(xm.nonschema, 'NO') as nonschema
      FROM dba_segments s
      JOIN dba_xml_tables xm
      ON s.owner = xm.owner
      AND s.segment_name = xm.table_name
      WHERE s.tablespace_name = '${tablespace}' ${exc_tbl_filter_xml_cont}
      and ( xm.storage_type <> 'BINARY' or nvl(xm.nonschema, 'NO') <> 'YES' )
    ) loop
    l_msg := l_msg || chr(10) ||
               rec.owner || '.' || rec.table_name ||
               ' has XMLType storage_type = ' || rec.storage_type ||
               ', nonschema = ' || rec.nonschema;
    end loop;  
    
    if length(l_msg) > 0 then
      l_msg :=
        'ERROR : Tablespace ${tablespace} has XMLType tables that are not non-schema Binary XML. ' ||
        'ADBS supports XMLType Data Pump migration only for non-schema Binary XML. ' ||
        'Failing tables:' || l_msg;
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line(l_msg);
      else
        raise_application_error(-20001, l_msg);
      end if;
    end if;

    -- check timezone is not Timpstamp with Local Time Zone (TSLTZ)
    select dbtimezone into l_timezone from dual;
    if (l_timezone != '+00:00') then
      l_msg := '';
      for rec in (
        select atc.owner,atc.table_name,atc.column_name from all_tab_columns atc
        join all_tables at on atc.table_name = at.table_name and atc.owner = at.owner
        where atc.data_type = 'TIMESTAMP WITH LOCAL TIME ZONE' and
        at.tablespace_name = '${tablespace}'
        order by atc.owner, atc.table_name, atc.column_name
      ) loop
        l_msg := l_msg || chr(10) ||
                 rec.owner || '.' || rec.table_name || '.' || rec.column_name;
      end loop;

      if length(l_msg) > 0 then
        l_msg := 'ERROR : Source database time zone is not UTC. DBTIMEZONE: ' || l_timezone ||
                 '. Tablespace "${tablespace}" contains TIMESTAMP WITH LOCAL TIME ZONE columns. ' ||
                 'Failing columns:' || l_msg;
        if ('${dry_run}' = 'TRUE') then
          dbms_output.put_line(l_msg);
        else
          raise_application_error(-20001, l_msg);
        end if;
      end if;
    end if;

    -- run transport charset check
    l_tts_charset_status := dbms_tts.transport_char_set_check(ts_list => '${tablespace}',
      target_db_char_set_name => 'AL32UTF8', target_db_nchar_set_name => 'AL16UTF16', err_msg => l_error_msg);
    if (l_tts_charset_status = false) then
        select property_value into l_nls_charset from database_properties where property_name = 'NLS_CHARACTERSET';
        select property_value into l_nls_ncharset from database_properties where property_name = 'NLS_NCHAR_CHARACTERSET';
        l_msg := 'Transport tablespace charset check failed for tablespace ${tablespace}: ' || l_error_msg;
        if ('${dry_run}' = 'TRUE') then
          dbms_output.put_line(l_msg);
        else
          raise_application_error(-20001, l_msg);
        end if;
    end if;
  exception
    when others then
      if ('${dry_run}' = 'TRUE') then
        dbms_output.put_line('ERROR : ' ||SQLERRM);
      else
        raise_application_error(-20001, 'ERROR : ' || SQLERRM);
      end if;
  end;
  /

      end if;
    end if;
  end;
  /

get_required_local_schemas:
  WITH local_users AS (
    SELECT username
    FROM dba_users
    WHERE ${l_user_list_clause}
  ),
  base_schemas AS (
    SELECT DISTINCT s.owner AS schema_name
    FROM dba_segments s
    WHERE s.tablespace_name IN (${ts_list})
    UNION
    SELECT DISTINCT t.owner AS schema_name
    FROM dba_tables t
    WHERE t.tablespace_name IN (${ts_list})
    UNION
    SELECT DISTINCT i.owner AS schema_name
    FROM dba_indexes i
    WHERE i.tablespace_name IN (${ts_list})
  ),
  trigger_schemas AS (
    SELECT tr.owner AS schema_name
    FROM dba_triggers tr
    JOIN dba_tables ta
      ON ta.owner = tr.table_owner
    AND ta.table_name = tr.table_name
    WHERE ta.tablespace_name IN (${ts_list})
  ),
  synonym_schemas AS (
    SELECT sy.owner AS schema_name
    FROM dba_synonyms sy
    JOIN dba_tables ta
      ON ta.owner = sy.table_owner
    AND ta.table_name = sy.table_name
    WHERE ta.tablespace_name IN (${ts_list})
  ),
  logical_schemas AS (
    SELECT o.owner AS schema_name
    FROM dba_dependencies d
    JOIN dba_objects o
      ON o.owner = d.owner
    AND o.object_name = d.name
    AND o.object_type = d.type
    JOIN base_schemas b
      ON b.schema_name = d.referenced_owner
    WHERE o.object_type IN (
      'VIEW',
      'PROCEDURE',
      'FUNCTION',
      'PACKAGE',
      'PACKAGE BODY',
      'TYPE',
      'TYPE BODY',
      'MATERIALIZED VIEW'
    )
  ),
  grant_schemas AS (
    SELECT p.grantee AS schema_name
    FROM dba_tab_privs p
    JOIN base_schemas b
      ON b.schema_name = p.owner
  ),
  candidate_schemas AS (
    SELECT schema_name FROM base_schemas
    UNION ALL
    SELECT schema_name FROM trigger_schemas
    UNION ALL
    SELECT schema_name FROM synonym_schemas
    UNION ALL
    SELECT schema_name FROM logical_schemas
    UNION ALL
    SELECT schema_name FROM grant_schemas
  )
  SELECT DISTINCT c.schema_name
  FROM candidate_schemas c
  JOIN local_users u
    ON u.username = c.schema_name
  ORDER BY c.schema_name;

logical_object_validation:
  -- DBA_SEGMENTS : physical segments
  -- we mandated all owners from dba_segments to be in schema_list

  -- TRIGGERS :
  -- Cross-schema triggers owned by excluded schemas are not transported.
  SELECT 'TRIGGER,' || tr.owner || '.' || tr.trigger_name
  FROM dba_triggers tr
  JOIN dba_tables ta
    ON ta.owner = tr.table_owner
  AND ta.table_name = tr.table_name
  WHERE ta.tablespace_name IN (${ts_list})
  AND tr.owner NOT IN (${sc_list});

  -- SYNONYMS :
  -- Cross-schema synonyms owned by excluded schemas are not transported.
  SELECT 'SYNONYM,' || sy.owner || '.' || sy.synonym_name
  FROM dba_synonyms sy
  JOIN dba_tables ta ON sy.table_name = ta.table_name AND sy.table_owner = ta.owner
  WHERE ta.tablespace_name IN (${ts_list})
    AND sy.owner NOT IN (${sc_list});

  -- LOGICAL OBJECTS :
  -- Logical objects owned by excluded schemas and referencing transported
  -- objects are considered out-of-scope and are not transported.
  -- Includes :
  -- VIEW, PROCEDURE, FUNCTION, PACKAGE, PACKAGE BODY,
  -- TYPE, TYPE BODY, MATERIALIZED VIEW
  SELECT DISTINCT 'OBJECT,' || o.owner || '.' || o.object_name || ',' || o.object_type AS OBJECT_NAME
  FROM dba_dependencies d
  JOIN dba_objects o
    ON o.owner = d.owner
  AND o.object_name = d.name
  AND o.object_type = d.type
  WHERE d.referenced_owner IN (${sc_list})
    AND o.object_type IN (
      'VIEW',
      'PROCEDURE',
      'FUNCTION',
      'PACKAGE',
      'PACKAGE BODY',
      'TYPE',
      'TYPE BODY',
      'MATERIALIZED VIEW'
    )
    AND o.owner NOT IN (${sc_list});

  -- OBJECT GRANTS :
  -- Grants to grantees outside sc_list are not transported.
  SELECT 'OBJECT_GRANT_FAIL,' || ' GRANT TO ' || grantee || ' on ' || owner || '.' || table_name || ' will fail (GRANTEE NOT TRANSPORTED)'
  FROM dba_tab_privs
  WHERE owner in (${sc_list})
    AND grantee in (
      SELECT username
      FROM dba_users
      WHERE username not in (${sc_list}) AND ${l_user_list_clause}
    );

tts_src_gather_data:
  declare
    l_platform_name v$database.platform_name%TYPE;
    l_platform_id v$database.platform_id%TYPE;
    l_current_scn v$database.current_scn%TYPE;
    l_product product_component_version.product%TYPE;
    l_version product_component_version.version%TYPE;
    l_version_full VARCHAR2(160);
    l_dst_version database_properties.property_value%TYPE;
    l_nls_charset database_properties.property_value%TYPE;
    l_nls_ncharset database_properties.property_value%TYPE;
    l_pdb_guid v$$${l_pdb_table}.${l_pdb_type}%TYPE;
    l_xmltype_exists varchar2(10);
    l_timezone varchar2(32);
    l_storage_size dba_data_files.bytes%TYPE;
    l_sf_additional_size dba_data_files.bytes%TYPE;
    l_bf_additional_size dba_data_files.bytes%TYPE;
    l_host_list varchar2(4000) := '';
    l_tbs_read_only varchar2(10);
    l_bigfile_list varchar2(30000) := '';
    l_smallfile_list varchar2(30000) := '';
    l_encrypted_list varchar2(30000) := '';
    l_unencrypted_list varchar2(30000) := '';
    l_role_list       clob := empty_clob();
    l_mview_schemas   clob := empty_clob();
    l_sched_cred_list clob := empty_clob();
    l_src_compatible  varchar2(160);
    l_server_type     varchar2(9);
    l_read_only_host_list varchar2(4000) := '';
  begin
    select ${l_pdb_type} into l_pdb_guid from v$$${l_pdb_table};
    select property_value into l_dst_version from database_properties where property_name = 'DST_PRIMARY_TT_VERSION';
    select platform_name, platform_id, current_scn into l_platform_name, l_platform_id, l_current_scn from v$database;
    select property_value into l_nls_charset from database_properties where property_name = 'NLS_CHARACTERSET';
    select property_value into l_nls_ncharset from database_properties where property_name = 'NLS_NCHAR_CHARACTERSET';
    select product, version into l_product, l_version from product_component_version where product like 'Oracle%Database%';
    BEGIN
      EXECUTE IMMEDIATE 'SELECT ${l_version_type} FROM product_component_version where product like ''Oracle%Database%''' INTO l_version_full;
      EXCEPTION
        WHEN OTHERS THEN
          l_version_full := l_version;
    END;

    select dbtimezone into l_timezone from dual;
    select (sum(bytes) / 1024 / 1024 / 1024) into l_storage_size from dba_data_files where tablespace_name in (${ts_list});
    
    -- smallfile tablespaces move require additional size
    select nvl(max(sum(df.bytes)/1024/1024/1024),0) into l_sf_additional_size
    from dba_data_files df, dba_tablespaces dt 
    where df.tablespace_name = dt.tablespace_name and 
    df.tablespace_name in (${ts_list}) and dt.bigfile = 'NO' 
    group by df.tablespace_name;

    -- bigfile tablespaces move require additional size
    select nvl(max(sum(df.bytes)/1024/1024/1024),0) into l_bf_additional_size
    from dba_data_files df, dba_tablespaces dt 
    where df.tablespace_name = dt.tablespace_name and 
    df.tablespace_name in (${ts_list}) and dt.bigfile = 'YES' 
    group by df.tablespace_name;
    
    -- Collect RAC instances where the source database/PDB is open READ WRITE.
    -- Used for RMAN channels on a primary database or snapshot standby.
    FOR rec IN (SELECT host_name, instance_name FROM gv$$${l_pdb_table} pdb, gv$instance inst
                  WHERE pdb.name = UPPER('${database_name}') 
                  AND pdb.open_mode = 'READ WRITE' 
                  AND pdb.inst_id = inst.inst_id) LOOP
          l_host_list := l_host_list || ';' || rec.host_name || ':' || rec.instance_name;
    END LOOP;
    l_host_list := SUBSTR(l_host_list, 2);

    -- Collect RAC instances where the source database/PDB is open READ ONLY.
    -- Used for RMAN channels when the source is a physical standby.
    FOR rec IN (SELECT host_name, instance_name FROM gv$$${l_pdb_table} pdb, gv$instance inst
                  WHERE pdb.name = UPPER('${database_name}') 
                  AND pdb.open_mode = 'READ ONLY' 
                  AND pdb.inst_id = inst.inst_id) LOOP
          l_read_only_host_list := l_read_only_host_list || ';' || rec.host_name || ':' || rec.instance_name;
    END LOOP;
    l_read_only_host_list := SUBSTR(l_read_only_host_list, 2);

    select decode(count(*), 0, 'true', 'false') into l_tbs_read_only from dba_tablespaces where status != 'READ ONLY' and tablespace_name in (${ts_list});

    FOR rec IN (SELECT tablespace_name, bigfile, encrypted FROM dba_tablespaces where tablespace_name IN (${ts_list})) LOOP
      IF rec.bigfile = 'YES' THEN
        l_bigfile_list := l_bigfile_list || ';' || rec.tablespace_name;
      ELSE
        l_smallfile_list := l_smallfile_list || ';' || rec.tablespace_name;
      END IF;

      IF rec.encrypted = 'YES' THEN
        l_encrypted_list := l_encrypted_list || ';' || rec.tablespace_name;
      ELSE
        l_unencrypted_list := l_unencrypted_list || ';' || rec.tablespace_name;
      END IF;
    END LOOP;
    
    IF LENGTH(l_bigfile_list) > 0 THEN
        l_bigfile_list := SUBSTR(l_bigfile_list, 2);
    END IF;
    IF LENGTH(l_smallfile_list) > 0 THEN
        l_smallfile_list := SUBSTR(l_smallfile_list, 2);
    END IF;
    IF LENGTH(l_encrypted_list) > 0 THEN
        l_encrypted_list := SUBSTR(l_encrypted_list, 2);
    END IF;
    IF LENGTH(l_unencrypted_list) > 0 THEN
        l_unencrypted_list := SUBSTR(l_unencrypted_list, 2);
    END IF;
    
    FOR rec IN (SELECT distinct grantee from dba_tab_privs where owner in (${sc_list}) and 
                grantee in (select role from dba_roles ${l_common_clause})) loop
      l_role_list := l_role_list || ';' || rec.grantee;
    END LOOP;
    IF DBMS_LOB.getlength(l_role_list) > 0 THEN
      l_role_list := DBMS_LOB.SUBSTR(l_role_list, DBMS_LOB.getlength(l_role_list) - 1, 2);
    END IF;

    FOR rec IN (SELECT distinct owner from dba_mviews where owner in (${sc_list})) loop
      l_mview_schemas := l_mview_schemas || ';' || rec.owner;
    END LOOP;
    IF DBMS_LOB.getlength(l_mview_schemas) > 0 THEN
      l_mview_schemas := DBMS_LOB.SUBSTR(l_mview_schemas, DBMS_LOB.getlength(l_mview_schemas) - 1, 2);
    END IF;

    FOR rec IN (SELECT credential_name from dba_scheduler_credentials where owner in (${sc_list})) loop
      l_sched_cred_list := l_sched_cred_list || ';' || rec.credential_name;
    END LOOP;
    IF DBMS_LOB.getlength(l_sched_cred_list) > 0 THEN
      l_sched_cred_list := DBMS_LOB.SUBSTR(l_sched_cred_list, DBMS_LOB.getlength(l_sched_cred_list) - 1, 2);
    END IF;

    select value into l_src_compatible from v$parameter where name = 'compatible';

    select server into l_server_type from v$session where sid = SYS_CONTEXT('USERENV','SID') and type = 'USER';

    dbms_output.put_line(l_pdb_guid       || ',' ||
                          l_dst_version    || ',' ||
                          l_platform_name  || ',' ||
                          l_platform_id    || ',' ||
                          l_current_scn    || ',' ||
                          l_nls_charset    || ',' ||
                          l_nls_ncharset   || ',' ||
                          l_product        || ',' ||
                          l_version        || ',' ||
                          l_version_full   || ',' ||
                          l_xmltype_exists || ',' ||
                          l_timezone       || ',' ||
                          l_storage_size   || ',' ||
                          l_host_list      || ',' ||
                          l_tbs_read_only  || ',' ||
                          l_bigfile_list   || ',' ||
                          l_smallfile_list || ',' ||
                          l_encrypted_list || ',' ||
                          l_unencrypted_list || ',' ||
                          l_sf_additional_size || ',' ||
                          l_bf_additional_size || ',' ||
                          l_role_list || ',' ||
                          l_mview_schemas || ',' ||
                          l_sched_cred_list || ',' ||
                          l_src_compatible || ',' ||
                          l_server_type || ',' ||
                          l_read_only_host_list);
  end;
  /

purge_dba_recyclebin:
  declare
    l_cnt number;
  begin
    if '${final_backup}' = 'TRUE' then
      select count(*) into l_cnt from dba_recyclebin
      where ts_name in (${ts_list});

      if l_cnt > 0 then
        raise_application_error(-20001,
          'Validation failed: Found BIN$ objects in one or more tablespaces from tbs list. ' ||
          'Please purge dba_recyclebin to avoid move failures at ADBS.'
        );
      end if;
    end if;
  end;
  /

validate_dvrealm_policies:
  declare
    l_dvconf_cnt number := 0;
    l_dvenabled_cnt number := 0;
    l_dv_roles_granted number := 0;
  begin
    select count(*) into l_dvconf_cnt from dba_dv_status WHERE name = 'DV_CONFIGURE_STATUS' and status = 'TRUE';
    select count(*) into l_dvenabled_cnt from dba_dv_status WHERE name = 'DV_ENABLE_STATUS' and status = 'TRUE';
    select 
      case 
        when count(distinct granted_role) = 2 then 1 else 0 
      end as dv_roles_granted
    into l_dv_roles_granted
    from dba_role_privs
    where granted_role in ('DV_OWNER','DV_ACCTMGR');
    dbms_output.put_line(l_dvconf_cnt || ',' || l_dvenabled_cnt || ',' || l_dv_roles_granted);
  end;
  /
